///$tab sub exe sub
Sub Exe_sub(sub_routine)
    Trace The routine $(sub_routine) just started;
    Let sub_start = Num(Now(1));
    Call $(sub_routine)
    Let sub_finish = Num(Now(1));
    Let sub_duration = Num(Round((sub_finish-sub_start)*86400),'#,##0');
    Trace The rutine $(sub_routine) completed, Time:$(sub_duration) seconds.;
End Sub

///$tab sub spaces
SUB get_spaces
TRACE ***Getting Spaces***;
  SET vParams = 'limit=100';
  
  [Spaces]:
  LOAD * INLINE [SpaceID];

  DO
  
    LET vAPICallCounter = $(vAPICallCounter) + 1;
  
   
    RestConnectorMasterTable:
    SQL SELECT 
        "__KEY_root",
        (SELECT 
            "id",
            "type",
            "ownerId",
            "tenantId",
            "name",
            "description",
            "createdAt",
            "createdBy",
            "updatedAt",
            "__KEY_data",
            "__FK_data",
            (SELECT 
                "__KEY_meta",
                "__FK_meta"
            FROM "meta" PK "__KEY_meta" FK "__FK_meta"),
            (SELECT 
                "__KEY_links",
                "__FK_links"
            FROM "links" PK "__KEY_links" FK "__FK_links")
        FROM "data" PK "__KEY_data" FK "__FK_data"),
        (SELECT 
            "__KEY_links_u0",
            "__FK_links_u0",
            (SELECT 
                "href" AS "href_u2",
                "__FK_next"
            FROM "next" FK "__FK_next")
        FROM "links" PK "__KEY_links_u0" FK "__FK_links_u0")
    FROM JSON (wrap on) "root" PK "__KEY_root"
    WITH CONNECTION (  
      URL "https://$(vu_tenant_fqdn)/api/v1/spaces?$(vParams)"
    );
    
    CONCATENATE (Spaces)
    LOAD	
      [id] AS SpaceID,
      Capitalize([type]) AS SpaceType,
      [ownerId] AS SpaceOwnerID,
      // [tenantId],
      [name] AS SpaceName,
      [description] AS SpaceDescription,
      TimeStamp([createdAt]) AS SpaceCreatedTime,
      [createdBy] AS SpaceCreatedBy
      // [updatedAt],
      // [__KEY_data],
      // [__FK_data] AS [__KEY_root]
    RESIDENT RestConnectorMasterTable
    WHERE NOT IsNull([__FK_data]);

    NextURL:
    LOAD	
    	[href_u2] AS NextURL
    RESIDENT RestConnectorMasterTable
    WHERE NOT IsNull([__FK_next]);

    DROP TABLE RestConnectorMasterTable;

    LET vParams = SubField(Peek('NextURL',0,'NextURL'),'?',-1);
    LET vNextURLRows = NoOfRows('NextURL');

    DROP TABLE NextURL;
    
  LOOP WHILE $(vNextURLRows)>0;

  AllSpacesMap:
  MAPPING LOAD DISTINCT
      SpaceID,
      1
  RESIDENT Spaces;
  
  Map_SpaceName_SpaceID:
  MAPPING LOAD DISTINCT
  	  Text(SpaceName) AS SpaceName,
      SpaceID
  RESIDENT Spaces;
  
  Map_SpaceID_SpaceName:
  MAPPING LOAD DISTINCT
  	  SpaceID,
  	  Text(SpaceName) AS SpaceName
  RESIDENT Spaces;
  
  Map_SpaceID_OwnerID:
  MAPPING LOAD DISTINCT
  	  SpaceID,
  	  SpaceOwnerID
  RESIDENT Spaces;

//aggr by day to get max reload per day
Round1:
LOAD 
  appId,
  appId & '|' & num(monthend(max(endTime))) as MonthKeyTemp,//link to associated reload details
  count(reloadId) as [# Reloads] ,
  max(endTime) as [Last Reload Time], 
  date(floor(max(endTime))) as [Last Reload Date], 
  MonthEnd(max(endTime)) as MonthEnd,
  max(loadExternalBytes) as MaxBytes,
  max(loadExternal_GB) as MaxGB,
  max(loadFilesBytes) as MaxFileBytes,
  max(loadFilesBytes_GB) as MaxFileBytes_GB,
  max(TotalBytes) as MaxTotalBytes,
  max(TotalBytes_GB) as MaxTotalBytes_GB,
  FirstSortedValue(loadExternalBytes, -endTime) AS [Last Reload Bytes],
  FirstSortedValue(loadExternal_GB, -endTime) AS [Last Reload GB],
  FirstSortedValue(loadFilesBytes, -endTime) AS [Last File Bytes],
  FirstSortedValue(loadFilesBytes_GB, -endTime) AS [Last File GB],
  FirstSortedValue(TotalBytes, -endTime) AS [Total Bytes],
  FirstSortedValue(TotalBytes_GB, -endTime) AS [Total GB]
 resident ReloadData  where success = 'True' group by appId, Date(floor((endTime))) order by endTime desc;
 
 NoConcatenate
 //aggr by month to get max and last reload per month
Round2:
Load
    appId,
    MonthKeyTemp,
    MonthEnd,
    sum([# Reloads]) as  [# Reloads],    
    Timestamp( Max([Last Reload Time]),'MMM DD YY hh:mm:ss') as [Last Reload Time], 
    max([Last Reload Date]) as [Last Reload Date], 
    max(MaxBytes) as MaxBytes,
    max(MaxGB) as MaxGB,
    max(MaxFileBytes) as MaxFileBytes,
    max(MaxFileBytes_GB) as MaxFileBytes_GB,
    max(MaxTotalBytes) as MaxTotalBytes,
    max(MaxTotalBytes_GB) as MaxTotalBytes_GB,
//     max([Last Reload GB]) as [Last Reload GB],
    FirstSortedValue(MaxBytes, -[Last Reload Time]) AS [Last Reload Bytes] ,//consider renameing to external
    FirstSortedValue(MaxGB, -[Last Reload Time]) AS [Last Reload GB] ,//consider renameing to external
    FirstSortedValue(MaxFileBytes, -[Last Reload Time]) AS [Last File Bytes] ,
    FirstSortedValue(MaxFileBytes_GB, -[Last Reload Time]) AS [Last File GB] ,
    FirstSortedValue(MaxTotalBytes, -[Last Reload Time]) AS [Last Total Bytes] ,
    FirstSortedValue(MaxTotalBytes_GB, -[Last Reload Time]) AS [Last Total GB]     
 resident Round1  group by appId, MonthKeyTemp, MonthEnd order by [Last Reload Time] desc;
 
 drop table Round1;
 
//Add columns for capacity consumption based on prior month or max of current month to date whichever is greater
MonthlySummary_Ranked:
//preceeding load to get current months usage based on prior months last reload value
Load MonthKeyTemp, MonthEnd,
    [Last Reload Date], MaxBytes, [Last Reload Bytes],
    PriorMonthReloadBytes,PriorReloadDate,MaxGB, [Last Reload GB],MaxFileBytes_GB,MaxFileBytes,PriorMonthFileBytes,PriorMonthFileBytes_GB,
    MaxTotalBytes, MaxTotalBytes_GB,PriorMonthTotalBytes,PriorMonthTotalBytes_GB,[Last File Bytes] ,[Last File GB],[Last Total Bytes] ,[Last Total GB] ,
    timestamp([Last Reload Time] ,'MMM DD YY hh:mm:ss') as [Last Reload Time], 
    if(PriorMonthReloadBytes > MaxBytes,'Prior Month Last Reload','In-Month Max Reload') as MaxConsumptionDesc,
    if(PriorMonthReloadBytes > MaxBytes,PriorMonthReloadBytes,MaxBytes) as MaxConsumption,
    if(PriorMonthReloadBytes > MaxBytes,PriorMonthReloadBytes_GB,MaxGB) as MaxConsumption_GB,
    if(PriorMonthFileBytes > MaxFileBytes,PriorMonthFileBytes,MaxFileBytes) as MaxConsumptionFiles,//this should be comparing the sum of both files and externalbytes?
    if(PriorMonthFileBytes > MaxFileBytes,PriorMonthFileBytes_GB,MaxFileBytes_GB) as MaxConsumptionFiles_GB,
    if(PriorMonthFileBytes > MaxTotalBytes,PriorMonthTotalBytes,MaxTotalBytes) as MaxConsumptionTotal,
    if(PriorMonthTotalBytes > MaxTotalBytes,PriorMonthTotalBytes_GB,MaxTotalBytes_GB) as MaxConsumptionTotal_GB,//this should be comparing the sum of both files and externalbytes?
    Window(RecNo(),appId, 'DESC', MonthEnd ) as MonthRank
// where Window(RecNo(),appId, 'DESC', MonthEnd ) < 3
;
LOAD 
    *,
    if(Previous(appId)=appId,date(floor(previous([Last Reload Date]))),date(floor([Last Reload Date]))) as PriorReloadDate,
    if(Previous(appId)=appId,previous([Last Reload Bytes]),[Last Reload Bytes]) as PriorMonthReloadBytes,
    if(Previous(appId)=appId,previous([Last Reload GB]),[Last Reload GB]) as PriorMonthReloadBytes_GB,
    if(Previous(appId)=appId,previous(MaxFileBytes),MaxFileBytes) as PriorMonthFileBytes,
    if(Previous(appId)=appId,previous(MaxFileBytes_GB),MaxFileBytes_GB) as PriorMonthFileBytes_GB,
    if(Previous(appId)=appId,previous(MaxTotalBytes),MaxTotalBytes) as PriorMonthTotalBytes,
    if(Previous(appId)=appId,previous(MaxTotalBytes_GB),MaxTotalBytes_GB) as PriorMonthTotalBytes_GB
    
RESIDENT Round2 ORDER BY appId, MonthEnd asc;

DROP TABLE Round2;
